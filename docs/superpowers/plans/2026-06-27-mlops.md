```sql
CREATE TABLE model_versions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  workspace_id UUID NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,

  -- 模型基本信息
  name VARCHAR(100) NOT NULL,
  version VARCHAR(50) NOT NULL,  -- semver: 1.0.0
  description TEXT,

  -- 制品关联
  artifact_id UUID REFERENCES artifacts(id),

  -- 训练信息
  trained_on_dataset_id UUID REFERENCES artifacts(id),  -- 数据集 artifact
  training_code_artifact_id UUID REFERENCES artifacts(id),  -- 代码 artifact
  training_config JSONB DEFAULT '{}',

  -- 评估指标
  metrics JSONB DEFAULT '{}',  -- { accuracy: 0.95, latency_ms: 12 }

  -- 上游版本（父模型）
  parent_model_version_id UUID REFERENCES model_versions(id),

  -- 部署状态
  is_production BOOLEAN DEFAULT false,
  production_deployment_id UUID REFERENCES deployments(id),

  created_by UUID REFERENCES users(id),
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_model_versions_workspace ON model_versions(workspace_id);
CREATE INDEX idx_model_versions_name ON model_versions(workspace_id, name);
CREATE INDEX idx_model_versions_production ON model_versions(workspace_id, is_production) WHERE is_production = true;
```

```typescript
export class WandbIntegration {
  private apiKey: string;

  constructor(apiKey: string) {
    this.apiKey = apiKey;
  }

  async getRunMetrics(runId: string) {
    const resp = await fetch(
      `https://api.wandb.ai/api/v1/runs/${runId}`,
      { headers: { Authorization: `Bearer ${this.apiKey}` } },
    );
    const data = await resp.json();
    return {
      summaryMetrics: data.summary_metrics,
      config: data.config,
    };
  }

  async listModels(entity: string, project: string) {
    const resp = await fetch(
      `https://api.wandb.ai/api/v1/models/${entity}/${project}`,
      { headers: { Authorization: `Bearer ${this.apiKey}` } },
    );
    return (await resp.json()).models;
  }
}
```
